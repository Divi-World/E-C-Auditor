### PATCH 4: Enterprise Proxy Health & Rotation (WAF Armor)
**The Problem:** Your WAF bypass uses `_get_proxies()`. If that function returns a static list of dead proxies, your `curl_cffi` requests will timeout, causing false negatives on high-value targets.

**The Fix:** Implement a self-healing `ProxyManager` that tests proxies before yielding them. Replace your current `_get_proxies()` logic with this:

```python
import random
import requests
from curl_cffi import requests as cffi_requests

class EnterpriseProxyManager:
    def __init__(self, proxy_list_path="proxies.txt"):
        self.raw_proxies = self._load_proxies(proxy_list_path)
        self.healthy_proxies = []

    def _load_proxies(self, path):
        try:
            with open(path, 'r') as f:
                return [line.strip() for line in f if line.strip()]
        except FileNotFoundError:
            return []

    def health_check(self, sample_size=10):
        """Tests a sample of proxies to ensure they are alive before use."""
        test_url = "https://httpbin.org/ip"
        candidates = random.sample(self.raw_proxies, min(sample_size, len(self.raw_proxies)))
        
        for proxy in candidates:
            try:
                # Use curl_cffi to test, matching your actual audit traffic
                r = cffi_requests.get(test_url, proxies={"http": proxy, "https": proxy}, 
                                      impersonate="chrome120", timeout=5)
                if r.status_code == 200:
                    self.healthy_proxies.append(proxy)
            except Exception:
                continue # Proxy is dead, discard it
        
        print(f"🛡️ Proxy Health Check: {len(self.healthy_proxies)}/{len(candidates)} proxies are active.")

    def get_proxy(self):
        if not self.healthy_proxies:
            self.health_check() # Auto-heal if pool is empty
        return random.choice(self.healthy_proxies) if self.healthy_proxies else None

# Initialize globally in site_audit.py
PROXY_MANAGER = EnterpriseProxyManager()

# Update your _get_proxies() function:
def _get_proxies():
    proxy = PROXY_MANAGER.get_proxy()
    return {"http": proxy, "https": proxy} if proxy else None
```
