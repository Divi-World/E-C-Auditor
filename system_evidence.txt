   137	def _goto_resilient(page, url: str, findings: dict = None, phase_name: str = "navigation") -> bool:
   138	    try:
   139	        page.goto(url, timeout=TIMEOUT_NAVIGATION, wait_until="domcontentloaded")
   140	        page.wait_for_timeout(4000)
   141	        return True
   142	    except PWTimeout:
   143	        if findings is not None:
   144	            findings["notes"] += f"{phase_name}_timeout_checking_dom. "
   145	        try:
   146	            state = page.evaluate("() => document.readyState")
   147	            if state in ["interactive", "complete"]:
   148	                if findings is not None:
   149	                    findings["notes"] += f"{phase_name}_dom_hydrated_post_timeout. "
   150	                return True
   151	        except Exception:
   152	            pass
   153	        return False
   154	    except Exception as e:
   155	        if findings is not None:
   156	            findings["notes"] += f"{phase_name}_fatal_error. "
   157	        return False

   172	def _extract_cwv_and_friction(page):
   173	    try:
   174	        return page.evaluate("""
   175	            () => {
   176	                return new Promise((resolve) => {
   177	                    let lcp = 0, cls = 0;
   178	                    try {
   179	                        const lcpEntries = performance.getEntriesByType('largest-contentful-paint');
   180	                        if (lcpEntries.length > 0) lcp = lcpEntries[lcpEntries.length - 1].startTime;
   181	                        const clsEntries = performance.getEntriesByType('layout-shift');
   182	                        cls = clsEntries.reduce((sum, e) => sum + (e.hadRecentInput ? 0 : e.value), 0);
   183	                    } catch(e) {}
   184	                    try {
   185	                        const lcpObs = new PerformanceObserver((list) => {
   186	                            const entries = list.getEntries();
   187	                            if (entries.length > 0) lcp = entries[entries.length - 1].startTime;
   188	                        });
   189	                        lcpObs.observe({ type: 'largest-contentful-paint', buffered: true });
   190	                    } catch(e) {}
   191	                    try {
   192	                        const clsObs = new PerformanceObserver((list) => {
   193	                            list.getEntries().forEach(entry => {
   194	                                if (!entry.hadRecentInput) cls += entry.value;
   195	                            });
   196	                        });
   197	                        clsObs.observe({ type: 'layout-shift', buffered: true });
   198	                    } catch(e) {}
   199	                    setTimeout(() => {
   200	                        let touch_target_ok = false;
   201	                        const atc = document.querySelector("button[name='add'], [data-add-to-cart], .single_add_to_cart_button");
   202	                        if (atc) {
   203	                            const r = atc.getBoundingClientRect();
   204	                            touch_target_ok = (r.width >= 32 && r.height >= 32);
   205	                        }
   206	                        resolve({ lcp: lcp > 0 ? Math.max(Math.round(lcp / 100) * 100, 100) : 0, cls: cls > 0 ? Math.max(Math.round(cls * 100) / 100, 0.01) : 0, touch_target_ok });
   207	                    }, 1500);
   208	                });
   209	            }
   210	        """)

   215	def find_a_product_url(page, domain: str) -> str | None:
   216	    try:
   217	        page.goto(f"https://{domain}/products.json?limit=10", timeout=TIMEOUT_HTTP_FALLBACK, wait_until="domcontentloaded")
   218	        data = _json.loads(page.inner_text("body"))
   219	        for prod in data.get("products", []):
   220	            if prod.get("handle"):
   221	                return f"https://{domain}/products/{prod['handle']}"
   222	    except Exception:
   223	        pass
   224	
   225	    discovery_urls = [
   226	        f"https://{domain}/collections/all", f"https://{domain}/collections",
   227	        f"https://{domain}/shop", f"https://{domain}/catalog",
   228	        f"https://{domain}/product-category", f"https://{domain}/products",
   229	        f"https://{domain}/store", f"https://{domain}/items",
   230	        f"https://{domain}/product", f"https://{domain}/all-products",
   231	        f"https://{domain}/shop/all", f"https://{domain}"
   232	    ]
   233	
   234	    product_url_pattern = re.compile(r'/(products?|p|shop|item|dp|catalog|buy)/[a-zA-Z0-9_\-]+/?$', re.I)
   235	    blacklist = ['cart', 'checkout', 'account', 'search', 'policies', 'blogs', 'pages', 'gift-card', 'login', 'register']
   236	
   237	    for url in discovery_urls:
   238	        if not _goto_resilient(page, url): continue
   239	        time.sleep(0.8)
   240	        dismiss_overlays(page)
   241	        links = page.evaluate("() => Array.from(document.querySelectorAll('a[href]')).map(a => a.getAttribute('href'))")
   242	        for href in links:
   243	            if not href: continue
   244	            clean_href = href.split('?')[0].split('#')[0]
   245	            is_product_path = any(p in clean_href.lower() for p in ['/product/', '/products/', '/p/', '/item/', '/dp/', '/buy/'])
   246	            if is_product_path or product_url_pattern.search(clean_href):
   247	                if any(bl in clean_href.lower() for bl in blacklist): continue
   248	                if '/product-category/' in clean_href.lower() or '/collections/' in clean_href.lower(): continue
   249	                return href if href.startswith("http") else f"https://{domain}{clean_href}"
   250	
   251	    try:
   252	        from curl_cffi import requests as cffi_requests
   253	        r = cffi_requests.get(f"https://{domain}", timeout=TIMEOUT_HTTP_FALLBACK/1000, impersonate="chrome120", proxies=_get_proxies())
   254	        if r.status_code == 200:
   255	            html_raw = r.text
   256	            matches = re.findall(r'href=["\'](https?://[^"\']*(?:/product/|/products/|/p/|/item/|/dp/|/shop/)[^"\']*)["\']', html_raw, re.I)
   257	            blacklist = ['cart', 'checkout', 'account', 'search', 'policies', 'blogs', 'pages', 'gift-card', 'login', 'register', 'category']
   258	            for m in matches:
   259	                clean = m.split('?')[0].split('#')[0].lower()
   260	                if not any(bl in clean for bl in blacklist): return m
   261	    except Exception:
   262	        pass
   263	
   264	    try:
   265	        page.goto(f"https://{domain}", timeout=TIMEOUT_NAVIGATION, wait_until="domcontentloaded")
   266	        time.sleep(0.8)
   267	        heuristic_url = page.evaluate("""
   268	            () => {
   269	                const btns = document.querySelectorAll('a, button');
   270	                for (const btn of btns) {
   271	                    const text = (btn.innerText || '').toLowerCase();
   272	                    const href = btn.getAttribute('href') || '';
   273	                    const is_buy_btn = text.includes('add to cart') || text.includes('buy now') || text.includes('subscribe') || text.includes('select plan') || text.includes('join now');
   274	                    if (is_buy_btn && href && href !== '#' && !href.includes('cart') && !href.includes('checkout')) {
   275	                        if (href.startsWith('http')) return href;
   276	                        return window.location.origin + href;
   277	                    }
   278	                }
   279	                return null;
   280	            }
   281	        """)
   282	        if heuristic_url: return heuristic_url
   283	    except Exception:
   284	        pass
   285	
   286	    # ENTERPRISE SITEMAP FALLBACK (Catches sites with non-standard product paths or WAF blocks)
   287	    try:
   288	        from curl_cffi import requests as cffi_requests
   289	        for sm_url in [f"https://{domain}/sitemap.xml", f"https://{domain}/sitemap_index.xml", f"https://{domain}/sitemap_products.xml"]:
   290	            try:
   291	                r = cffi_requests.get(sm_url, timeout=5, impersonate="chrome120", proxies=_get_proxies())
   292	                if r.status_code == 200:
   293	                    match = re.search(r'<loc>(https?://[^<]+/(?:products?|p|item|dp|shop)/[^<]+)</loc>', r.text, re.I)
   294	                    if match:
   295	                        url = match.group(1).split('?')[0]
   296	                        if not any(bl in url.lower() for bl in ['cart', 'checkout', 'account', 'search', 'policy']):
   297	                            return url
   298	            except Exception: continue
   299	    except Exception: pass
   300	
   301	    return None

   313	        time.sleep(0.8)

   315	            () => {

   321	                const has_payment_trust = payment_icons.length > 0 || footer.includes('secure checkout') || footer.includes('ssl');
   322	                return { has_free_shipping, has_returns, has_payment_trust };
   323	            }
   324	        """)
   325	        if footer_loaded and not hp_data.get('has_free_shipping') and not hp_data.get('has_returns'):
   326	            try:

   328	                    "document.body && document.body.innerText.length > 500",
   329	                    timeout=4000
   330	                )
   331	            except Exception:
   332	                pass
   333	            time.sleep(1.0)

   335	                () => {
   336	                    const html = document.body ? document.body.innerText.toLowerCase() : '';
   337	                    const footer = document.querySelector('footer') ? document.querySelector('footer').innerText.toLowerCase() : html;

   385	            if coll_data.get('has_grid') and coll_data.get('cards_with_price', 0) < coll_data.get('total_cards', 1) * 0.5:
   386	                findings["issues"].append({

   396	                    "evidence": "No filter/facet DOM detected on the product listing page.",
   397	                    "business_impact": "Users cannot narrow down large catalogs, leading to decision paralysis and bounce.",
   398	                    "fix": "Implement faceted navigation for key attributes (Price, Size, Color, Brand)."

    23	def opportunity_score(findings: dict) -> float:
    24	    """Enterprise Weighted Scoring Matrix (Baymard Impact Model)"""
    25	    if not findings or "issues" not in findings: return 10.0
    26	    score = 100.0
    27	    weights = {"high": 12.0, "medium": 3.5, "low": 0.5}
    28	    conf_mult = {"VERIFIED": 1.0, "high": 0.9, "medium": 0.6, "low": 0.3}
    29	    seen_codes = set()
    30	    for issue in findings.get("issues", []):
    31	        code = issue.get("code", "")
    32	        if code in seen_codes: continue
    33	        seen_codes.add(code)
    34	        sev = issue.get("severity", "low")
    35	        conf = issue.get("confidence", "medium")
    36	        base_penalty = weights.get(sev, 2.0)
    37	        multiplier = conf_mult.get(conf, 0.5)
    38	        if code in ["forced_account_creation", "checkout_hidden_fees_detected", "slow_ttfb_server_health"]:
    39	            base_penalty = 25.0
    40	        score -= (base_penalty * multiplier)
    41	    return max(1.0, round(score / 10.0, 1))  # Floor at 1.0

    # Check 1: no image URLs sampled as product/entity pages
    for issue in r.get("issues", []):
        for url in issue.get("affected_urls", []):
            if any(url.lower().split('?')[0].endswith(ext) for ext in IMAGE_EXTENSIONS):
                problems.append(f"image URL in affected_urls: {url}")

    # Check 2: capabilities PASS shouldn't coexist with a failure issue for same protocol
    caps = r.get("agentic_capabilities", {})
    for issue in r.get("issues", []):
        if issue["code"] in ("mcp_handshake_failed", "ucp_handshake_failed"):
            if caps.get("MCP") == "PASS" or caps.get("UCP") == "PASS":
                problems.append(f"contradiction: {issue['code']} present but capability shows PASS")

    # Check 3: unmeasured dimension should be null, not a number
    for dim, measured in r.get("dimensions_measured", {}).items():
        if not measured and r["dimensions"].get(dim) not in (None,):
            problems.append(f"{dim} marked unmeasured but has a numeric value: {r['dimensions'][dim]}")

    # Check 4: platform_detected always present
    if "platform_detected" not in r:
        problems.append("platform_detected missing entirely")

    status = "OK" if not problems else "ISSUES: " + "; ".join(problems)
    print(f"{domain:20} {status}")
